\begin{parts}

\part
رابطه به‌روزرسانی در \lr{Value Iteration} برابر است با:
\[
V_{k+1}(s)=
\max_a
\sum_{s'}T(s'|s,a)\left(R(s,a,s')+\gamma V_k(s')\right),
\qquad
\gamma=0.9
\]
در ابتدا:
\[
V_0(s)=0
\]
حالت‌های \lr{terminal} مقدار صفر دارند:
\[
V(3,1)=0,\qquad V(3,2)=0
\]

\textbf{مرحله اول:}

فقط دو حالت می‌توانند در یک حرکت به خانه با پاداش مثبت برسند:
\[
V_1(2,2)=0.9(10)=9,\qquad
V_1(3,3)=0.9(10)=9
\]
پس:
\[
V_1(1,1)=0,\quad
V_1(1,2)=0,\quad
V_1(1,3)=0,\quad
V_1(2,1)=0
\]
\[
V_1(2,2)=9,\quad
V_1(2,3)=0,\quad
V_1(3,3)=9
\]

\textbf{مرحله دوم:}
\[
V_2(1,1)=0
\]
\[
V_2(1,2)=0.9\left(0.9V_1(2,2)\right)=0.9(0.9\times 9)=7.29
\]
\[
V_2(1,3)=0
\]
\[
V_2(2,1)=0.9\left(0.9V_1(2,2)\right)+0.05(-10)
=7.29-0.5=6.79
\]
\[
V_2(2,2)=0.9(10)=9
\]
\[
V_2(2,3)
=
0.9\left(0.9V_1(3,3)\right)
+
0.05\left(0.9V_1(2,2)\right)
=
7.29+0.405=7.695
\]
\[
V_2(3,3)
=
0.9(10)+0.05\left(0.9V_1(3,3)\right)
=
9+0.405=9.405
\]

پس:
\[
V_2(1,1)=0,\quad
V_2(1,2)=7.29,\quad
V_2(1,3)=0
\]
\[
V_2(2,1)=6.79,\quad
V_2(2,2)=9,\quad
V_2(2,3)=7.695,\quad
V_2(3,3)=9.405
\]

\part
در \lr{TD Learning} داریم:
\[
V(s)\leftarrow V(s)+\alpha\left(r+\gamma V(s')-V(s)\right),
\qquad
\alpha=0.1,\quad \gamma=0.9
\]
در ابتدا همه مقدارها صفر هستند.

\textbf{مسیر اول:}
\[
(1,1)\to(1,2)\to(1,3)
\]
در این مسیر \lr{reward}ای دریافت نمی‌شود:
\[
V(1,1)\leftarrow 0+0.1\left(0+0.9V(1,2)-0\right)=0
\]
\[
V(1,2)\leftarrow 0+0.1\left(0+0.9V(1,3)-0\right)=0
\]

\textbf{مسیر دوم:}
\[
(1,1)\to(1,2)\to(2,2)\to(2,3)
\]
باز هم \lr{reward}ای دریافت نمی‌شود:
\[
V(1,1)\leftarrow 0+0.1\left(0+0.9V(1,2)-0\right)=0
\]
\[
V(1,2)\leftarrow 0+0.1\left(0+0.9V(2,2)-0\right)=0
\]
\[
V(2,2)\leftarrow 0+0.1\left(0+0.9V(2,3)-0\right)=0
\]

پس:
\[
V(1,1)=0,\qquad V(1,2)=0,\qquad V(2,2)=0
\]
و بقیه مقدارها نیز صفر می‌مانند.

\part
در \lr{SARSA} داریم:
\[
Q(s,a)\leftarrow Q(s,a)+\alpha\left(r+\gamma Q(s',a')-Q(s,a)\right),
\qquad
\alpha=0.1,\quad \gamma=0.9
\]
در ابتدا:
\[
Q_0(s,a)=0
\]
کنش راست را با \(a_R\) نشان می‌دهیم.

\textbf{مسیر اول:}
\[
(1,1)\to(1,2)\to(1,3)
\]
\[
Q((1,1),a_R)
\leftarrow
0+0.1\left(0+0.9Q((1,2),a_R)-0\right)=0
\]
\[
Q((1,2),a_R)
\leftarrow
0+0.1\left(0+0.9Q((1,3),a_R)-0\right)=0
\]

\textbf{مسیر دوم:}
\[
(1,1)\to(1,2)\to(2,2)\to(2,3)
\]
\[
Q((1,1),a_R)
\leftarrow
0+0.1\left(0+0.9Q((1,2),a_R)-0\right)=0
\]
\[
Q((1,2),a_R)
\leftarrow
0+0.1\left(0+0.9Q((2,2),a_R)-0\right)=0
\]
\[
Q((2,2),a_R)
\leftarrow
0+0.1\left(0+0.9Q((2,3),a_R)-0\right)=0
\]

پس:
\[
Q((1,1),a_R)=0,\qquad
Q((1,2),a_R)=0,\qquad
Q((2,2),a_R)=0
\]

\part
در \lr{Q-learning} داریم:
\[
Q(s,a)\leftarrow
Q(s,a)+\alpha\left(r+\gamma\max_{a'}Q(s',a')-Q(s,a)\right)
\]
تفاوتش با \lr{SARSA} این است که در \lr{SARSA} از \lr{action} واقعی بعدی استفاده می‌شود، ولی در \lr{Q-learning} بهترین \lr{action} ممکن در حالت بعدی را می‌گیریم.

چون مقدار اولیه همه \(Q\)ها صفر است و در این دو مسیر \lr{reward} غیرصفری نداریم، همه آپدیت‌ها صفر می‌شوند:
\[
Q((1,1),a_R)=0,\qquad
Q((1,2),a_R)=0,\qquad
Q((2,2),a_R)=0
\]

پس در این مثال مقدار عددی \lr{SARSA} و \lr{Q-learning} یکی می‌شود، ولی از نظر مفهومی یکی نیستند.

\part
در \lr{Direct Evaluation} برای هر حالتی که در مسیر دیده شده، \lr{return} تخفیف‌خورده را حساب می‌کنیم:
\[
G_t=\sum_{k=0}^{T-t-1}\gamma^k r_{t+k+1},
\qquad
\gamma=0.9
\]

\textbf{مسیر اول:}
\[
(1,1)\to(1,2)\to(1,3)
\]
چون \lr{reward} نداریم:
\[
G(1,1)=0,\qquad G(1,2)=0
\]

\textbf{مسیر دوم:}
\[
(1,1)\to(1,2)\to(2,2)\to(2,3)
\]
باز هم \lr{reward} نداریم:
\[
G(1,1)=0,\qquad G(1,2)=0,\qquad G(2,2)=0
\]

پس:
\[
\hat{V}(1,1)=\frac{0+0}{2}=0
\]
\[
\hat{V}(1,2)=\frac{0+0}{2}=0
\]
\[
\hat{V}(2,2)=0
\]

حالت‌های \((1,3)\) و \((2,3)\) انتهای مسیرهای مشاهده‌شده هستند، پس \lr{return} قابل‌استفاده‌ای برایشان نداریم. حالت‌هایی مثل \((2,1)\) و \((3,3)\) هم دیده نشده‌اند.

از نظر \lr{sample efficiency}، روش \lr{Direct Evaluation} ضعیف‌تر از \lr{TD Learning} است، چون باید ادامه مسیر را ببیند. ولی \lr{TD Learning} با هر گام می‌تواند آپدیت انجام دهد و اطلاعات را سریع‌تر پخش کند.

\end{parts}
